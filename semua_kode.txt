class Command(BaseCommand):
    help = 'Buat akun admin: python manage.py buat_admin budi "Pak Budi" kataSandi'

    def add_arguments(self, parser):
        parser.add_argument("username")
        parser.add_argument("nama")
        parser.add_argument("sandi")

    def handle(self, *args, **o):
        if len(o["sandi"]) < 6:
            raise CommandError("Kata sandi minimal 6 karakter.")
        if User.objects.filter(username=o["username"]).exists():
            raise CommandError(f"Username {o['username']} sudah dipakai.")
        u = User.objects.create_user(o["username"], password=o["sandi"], first_name=o["nama"])
        Profile.objects.create(user=u, role="admin")
        self.stdout.write(f"Admin {u.username} dibuat. Masuk di /login/ untuk membuat kelas dan akun guru.")

HOME = {"murid": "/murid/", "guru": "/guru/", "admin": "/admin/"}


def get_role(user):
    """'murid' | 'guru' | 'admin' | None. Tiga role: murid, guru, dan admin."""
    if not user.is_authenticated:
        return None
    profile = getattr(user, "profile", None)
    return profile.role if profile else None


def role_required(*roles):
    def deco(view):
        @wraps(view)
        @login_required
        def wrapped(request, *a, **kw):
            role = get_role(request.user)
            return view(request, *a, **kw) if role in roles else redirect(HOME.get(role, "/login/"))
        return wrapped
    return deco

def login_view(request):
    role = get_role(request.user)
    if role:
        return redirect(HOME[role])
    ctx = {}
    kode = (request.POST.get("kode") or request.GET.get("kode") or "").strip().upper()
    if kode:  # alur murid (F-01): kode kelas -> pilih avatar, tanpa kata sandi
        kelas = Kelas.objects.filter(kode=kode).first()
        ctx["kode"] = kode
        if not kelas:
            ctx["error"] = "Kode kelas tidak ditemukan."
        elif request.POST.get("siswa"):
            s = kelas.siswa.select_related("user").filter(id=request.POST["siswa"]).first()
            if s:
                login(request, s.user, backend="django.contrib.auth.backends.ModelBackend")
                return redirect(HOME["murid"])
        else:
            ctx["kelas"], ctx["daftar"] = kelas, kelas.siswa.select_related("user")
    elif request.method == "POST":  # alur guru/admin: username + kata sandi
        u = authenticate(request, username=request.POST.get("username"), password=request.POST.get("password"))
        if u and get_role(u):
            login(request, u)
            return redirect(HOME[get_role(u)])
        ctx["error"] = "Username atau kata sandi salah."
    return render(request, "accounts/login.html", ctx)


def logout_view(request):
    logout(request)
    return redirect("/login/")

def buat_user(username, nama, role, sandi=None):
    u, baru = User.objects.get_or_create(username=username, defaults={"first_name": nama})
    if baru:
        u.set_password(sandi) if sandi else u.set_unusable_password()
        u.save()
    Profile.objects.get_or_create(user=u, defaults={"role": role})
    return u

def _ctx(request, aktif, **kw):
    return {"admin": request.user, "aktif": aktif, **kw}


def _semua_kelas():
    """Semua kelas dengan jumlah siswa dan daftar siswanya (untuk ditampilkan/diciutkan per kelas)."""
    return (
        Kelas.objects.select_related("wali_kelas")
        .annotate(jumlah_siswa=Count("siswa"))
        .prefetch_related("siswa__user")
        .order_by("nama")
    )


def _semua_guru():
    """Semua akun guru, masing-masing disisipi .kelas (kelas yang diampu, atau None)."""
    guru = list(User.objects.filter(profile__role="guru").order_by("username"))
    kelas_by_guru = {k.wali_kelas_id: k for k in Kelas.objects.exclude(wali_kelas__isnull=True)}

    for g in guru:
        g.kelas = kelas_by_guru.get(g.id)

    return guru


@role_required("admin")
def admin_kelas(request):
    if request.method == "POST":
        aksi, p = request.POST.get("aksi"), request.POST
        try:
            if aksi == "buat_kelas":
                nama = p["nama"].strip()[:20]
                if not nama:
                    raise ValueError
                buat_kelas(None, nama)
                messages.success(request, f"Kelas “{nama}” dibuat. Tambahkan siswa dan bagikan kode kelasnya ke guru.")
            elif aksi == "tambah_siswa":
                kelas = get_object_or_404(Kelas, id=p["kelas"])
                nama = [x.strip()[:40] for x in p["nama"].splitlines() if x.strip()]
                if not nama:
                    raise ValueError
                for x in nama:
                    tambah_siswa(kelas, x)
                messages.success(request, f"{len(nama)} siswa ditambahkan ke kelas {kelas.nama}. Murid tinggal masuk pakai kode kelas, lalu pilih namanya sendiri.")
            elif aksi == "hapus_siswa":
                kelas = get_object_or_404(Kelas, id=p["kelas"])
                get_object_or_404(kelas.siswa.all(), id=p["id"]).user.delete()
                messages.success(request, "Siswa dihapus beserta datanya.")
            elif aksi == "ganti_kode":
                kelas = get_object_or_404(Kelas, id=p["kelas"])
                kelas.kode = kode_kelas_baru(kelas.nama)
                kelas.save()
                messages.success(request, f"Kode kelas {kelas.nama} diperbarui: {kelas.kode}. Kartu masuk lama perlu dicetak ulang.")
        except IntegrityError:
            messages.error(request, "Nama kelas itu sudah dipakai.")
        except (ValueError, KeyError):
            messages.error(request, "Isian tidak valid.")
        return redirect("/admin/")

    return render(request, "belajar/admin_kelas.html", _ctx(request, "kelas", daftar_kelas=_semua_kelas()))


@role_required("admin")

            elif aksi == "pindah_kelas":

def sinkron_taksonomi(sender=None, **kwargs):
    """
    Sinkronkan kompetensi dan aktivitas ke database.

    Kompetensi dan aktivitas lama tidak dihapus otomatis.
    Riwayat jawaban, penguasaan, serta misi siswa harus
    tetap terjaga.
    """

    from . import engine
    from .models import Aktivitas, Kompetensi

    # --------------------------------------------------
    # 1. Sinkronkan seluruh kompetensi
    # --------------------------------------------------

    for urutan, item in enumerate(engine.TAKSONOMI):
        (
            kode,
            ranah,
            nama,
            judul,
            cakupan,
            _contoh,
        ) = item

        Kompetensi.objects.update_or_create(
            kode=kode,
            defaults={
                "ranah": ranah,
                "nama": nama,
                "judul": judul,
                "cakupan": cakupan,
                "urutan": urutan,
            },
        )

    # --------------------------------------------------
    # 2. Ambil kompetensi yang sudah tersinkron
    # --------------------------------------------------

    kompetensi_map = {
        kompetensi.kode: kompetensi
        for kompetensi in Kompetensi.objects.all()
    }

    # --------------------------------------------------
    # 3. Pastikan aktivitas resmi tersedia
    #
    # Aktivitas yang sudah ada akan diperbarui deskripsi,
    # level, dan durasinya tanpa mengubah status validasi.
    #
    # Aktivitas baru mengikuti status validasi awal pada
    # BANK_AWAL.
    # --------------------------------------------------

    for (
        judul,
        kode,
        level_min,
        level_maks,
        menit,
        deskripsi,
        validasi_awal,
    ) in BANK_AWAL:

        kompetensi = kompetensi_map.get(kode)

        # Jangan membuat aktivitas jika kompetensinya
        # tidak ditemukan di engine.TAKSONOMI.
        if kompetensi is None:
            continue

        aktivitas = (
            Aktivitas.objects.filter(
                kompetensi=kompetensi,
                judul=judul,
            )
            .order_by("id")
            .first()
        )

        defaults = {
            "level_min": level_min,
            "level_maks": level_maks,
            "menit": menit,
            "deskripsi": deskripsi,
        }

        if aktivitas is None:
            # Aktivitas baru dibuat sesuai status validasi
            # awal yang ditetapkan dalam BANK_AWAL.
            Aktivitas.objects.create(
                kompetensi=kompetensi,
                judul=judul,
                tervalidasi=validasi_awal,
                **defaults,
            )

        else:
            # Perbarui informasi aktivitas tanpa mengubah
            # status validasi guru.
            for field, value in defaults.items():
                setattr(aktivitas, field, value)

            aktivitas.save(
                update_fields=list(defaults.keys())
            )

    # Sengaja tidak menghapus aktivitas di luar BANK_AWAL.
    # Aktivitas buatan guru, aktivitas terdahulu, dan
    # catatan terkait tidak boleh terhapus otomatis.

class GameCorsMiddleware:
    """Izinkan game/quiz di origin lain memanggil /api/game/. Auth pakai Bearer token (bukan cookie).
    LENTERA_GAME_ORIGIN = "*" (dev) atau daftar origin dipisah koma (produksi)."""

    def __init__(self, get_response):
        self.get_response = get_response

    def __call__(self, request):
        api = request.path.startswith("/api/game/")
        resp = HttpResponse(status=204) if api and request.method == "OPTIONS" else self.get_response(request)
        if api:
            izin = [o.strip() for o in settings.LENTERA_GAME_ORIGIN.split(",")]
            asal = request.headers.get("Origin", "")
            if "*" in izin:
                resp["Access-Control-Allow-Origin"] = "*"
            elif asal in izin:
                resp["Access-Control-Allow-Origin"] = asal
                resp["Vary"] = "Origin"
            resp["Access-Control-Allow-Headers"] = "Authorization, Content-Type"
            resp["Access-Control-Allow-Methods"] = "POST, OPTIONS"
        return resp

def _angka_valid(nilai):
    """Memastikan nilai numerik dan bukan NaN/infinity."""
    try:
        angka = float(nilai)
    except (TypeError, ValueError, OverflowError):
        return None

    if not isfinite(angka):
        return None

    return angka


def batasi_peluang(p):
    """Memastikan peluang selalu berada pada interval 0 sampai 1."""
    angka = _angka_valid(p)

    if angka is None:
        return P_AWAL

    return max(0.0, min(1.0, angka))


def fase(level):
    """
    Level 1-2 = Fase A.
    Level 3-5 = Fase B.
    """
    level = int(level)

    if not LEVEL_MIN <= level <= LEVEL_MAKS:
        raise ValueError("Level harus berada pada 1 sampai 5.")

    return "A" if level <= 2 else "B"


def status(p):
    """
    Kategori penguasaan:
    - Mulai: P(L) < 0.40
    - Berkembang: 0.40 <= P(L) < 0.80
    - Cakap: P(L) >= 0.80
    """
    p = batasi_peluang(p)

    if p >= AMBANG_CAKAP:
        return "Cakap"

    if p >= AMBANG_MULAI:
        return "Berkembang"

    return "Mulai"


# ==========================================================
# 7. BAYESIAN KNOWLEDGE TRACING
# ==========================================================

def update_bkt(p, benar):
    """
    Perbarui estimasi penguasaan setelah satu jawaban.

    p = peluang siswa menguasai kompetensi sebelum jawaban.
    benar = True jika jawaban benar, False jika salah.
    """
    p = batasi_peluang(p)

    if not isinstance(benar, bool):
        benar = bool(benar)

    if benar:
        pembilang = p * (1 - S)
        penyebut = (
            p * (1 - S)
            + (1 - p) * G
        )
    else:
        pembilang = p * S
        penyebut = (
            p * S
            + (1 - p) * (1 - G)
        )

    if penyebut <= 0:
        peluang_observasi = p
    else:
        peluang_observasi = pembilang / penyebut

    peluang_belajar = (
        peluang_observasi
        + (1 - peluang_observasi) * T
    )

    return round(
        batasi_peluang(peluang_belajar),
        6,
    )


# ==========================================================
# 8. PENYESUAIAN LEVEL
# ==========================================================

def atur_level(level, p, salah_beruntun):
    """
    Menyesuaikan level berdasarkan penguasaan.

    Aturan:
    - Naik satu level jika P(L) >= 0.95.
    - Turun satu level jika salah tiga kali berturut-turut.
    - Level tetap berada pada rentang 1 hingga 5.

    Return:
        (level_baru, p_l_baru, salah_beruntun_baru)
    """
    level = max(
        LEVEL_MIN,
        min(LEVEL_MAKS, int(level)),
    )

    p = batasi_peluang(p)

    salah_beruntun = max(
        0,
        int(salah_beruntun),
    )

    if salah_beruntun >= 3:
        level_baru = max(
            LEVEL_MIN,
            level - 1,
        )
        return level_baru, p, 0

    if p >= AMBANG_NAIK_LEVEL and level < LEVEL_MAKS:
        return (
            level + 1,
            AMBANG_CAKAP,
            0,
        )

    return (
        level,
        p,
        salah_beruntun,
    )


# ==========================================================
# 9. ESTIMASI PELUANG BENAR
# ==========================================================

def peluang_benar(p, level_siswa, level_item):
    """
    Estimasi peluang menjawab benar berdasarkan penguasaan
    dan perbedaan level soal.

    Nilai minimum 0.03 dan maksimum 0.97 menghindari
    peluang ekstrem.
    """
    p = batasi_peluang(p)

    level_siswa = max(
        LEVEL_MIN,
        min(LEVEL_MAKS, int(level_siswa)),
    )

    level_item = max(
        LEVEL_MIN,
        min(LEVEL_MAKS, int(level_item)),
    )

    hasil = (
        p * (1 - S)
        + (1 - p) * G
        - 0.15 * (level_item - level_siswa)
    )

    return max(
        0.03,
        min(0.97, hasil),
    )

# 10. PEMILIHAN BUTIR SOAL
# ==========================================================

def pilih_item(kandidat, p_l, level, riwayat=()):
    """
    Memilih soal dengan estimasi peluang benar mendekati 70%.

    Format kandidat:
        [
            {"id": "q1", "level": 1},
            {"id": "q2", "level": 2},
        ]

    Soal di lima posisi riwayat terakhir dihindari jika
    masih ada kandidat lain.

    Return:
        Satu kandidat, atau None bila tidak ada kandidat valid.
    """
    if not kandidat:
        return None

    kandidat_valid = []

    for item in kandidat:
        if not isinstance(item, dict):
            continue

        if not item.get("id"):
            continue

        try:
            level_item = int(item["level"])
        except (KeyError, TypeError, ValueError):
            continue

        if not LEVEL_MIN <= level_item <= LEVEL_MAKS:
            continue

        kandidat_valid.append(item)

    if not kandidat_valid:
        return None

    riwayat_terakhir = set(list(riwayat)[-5:])

    pool = [
        item
        for item in kandidat_valid
        if item["id"] not in riwayat_terakhir
    ]

    if not pool:
        pool = kandidat_valid

    return min(
        pool,
        key=lambda item: abs(
            peluang_benar(
                p_l,
                level,
                item["level"],
            ) - 0.70
        ),
    )

# 11. LEVEL AWAL DARI ASESMEN DIAGNOSTIK
# ==========================================================

def level_dari_skor(benar, total):
    """
    Konversi skor asesmen awal ke:
        (level_awal, P(L)_awal)

    Aturan:
        rasio = benar / total
        level = min(5, 1 + floor(rasio * 5))
        P(L)  = 0.10 + 0.75 * rasio
    """
    try:
        benar = int(benar)
        total = int(total)
    except (TypeError, ValueError):
        raise ValueError(
            "Jumlah benar dan total harus berupa bilangan bulat."
        )

    if total <= 0:
        raise ValueError(
            "Jumlah soal harus lebih besar dari nol."
        )

    if benar < 0 or benar > total:
        raise ValueError(
            "Jumlah jawaban benar harus berada antara 0 dan total."
        )

    rasio = benar / total

    level_awal = min(
        LEVEL_MAKS,
        1 + int(rasio * 5),
    )

    p_awal = round(
        0.10 + 0.75 * rasio,
        2,
    )

    return level_awal, p_awal

def _jumlah_soal_wajib(misi):
    """
    Semua kompetensi literasi menggunakan Game 1.
    Semua kompetensi numerasi menggunakan Game 2.

    Setiap level menggunakan 5 soal asli dari game
    yang sesuai. Tidak ada penambahan bank soal baru.

    Jumlah soal dapat diatur melalui
    LENTERA_GAME_REQUIRED_ITEMS di settings.py.
    """
    kode = misi.kompetensi.kode
    level = int(misi.level)

    # Level permainan yang didukung adalah 1 sampai 5.
    if level < 1 or level > 5:
        return 0

    # Kompetensi harus dipetakan ke salah satu dari dua game.
    game_map = getattr(
        settings,
        "LENTERA_GAMES",
        {},
    )

    if kode not in game_map:
        return 0

    # Hanya menerima ranah yang digunakan dashboard.
    if misi.kompetensi.ranah not in ("literasi", "numerasi"):
        return 0

    # Konfigurasi opsional untuk jumlah soal berbeda.
    konfigurasi = getattr(
        settings,
        "LENTERA_GAME_REQUIRED_ITEMS",
        {},
    )

    override = (
        konfigurasi.get(kode)
        if isinstance(konfigurasi, dict)
        else None
    )

    if isinstance(override, dict):
        try:
            jumlah = int(
                override.get(
                    level,
                    override.get(str(level), 5),
                )
            )
        except (ValueError, TypeError):
            raise ValueError(
                f"Jumlah soal {kode} level {level} tidak valid."
            )

        return max(0, jumlah)

    if override is not None:
        try:
            jumlah = int(override)
        except (ValueError, TypeError):
            raise ValueError(
                f"Jumlah soal {kode} tidak valid."
            )

        return max(0, jumlah)

    # Default: 5 soal untuk setiap level kompetensi.
    return 5

@role_required("murid")
def mulai_game(request, misi_id):
    misi = get_object_or_404(
        Misi.objects.select_related("siswa", "kompetensi"),
        id=misi_id,
        siswa=request.user.siswa,
        status="baru",
    )

    # Jangan membuka kompetensi-level yang belum punya soal.
    jumlah_wajib = _jumlah_soal_wajib(misi)

    if jumlah_wajib <= 0:
        return JsonResponse(
            {
                "error": (
                    f"Belum ada soal untuk {misi.kompetensi.kode} "
                    f"pada Level {misi.level}. "
                    "Pilih aktivitas lain yang tersedia."
                )
            },
            status=409,
        )

    # Hanya Game 1 atau Game 2 yang boleh dibuka.
    url_game = settings.LENTERA_GAMES.get(
        misi.kompetensi.kode
    )

    if not url_game:
        return JsonResponse(
            {
                "error": (
                    f"Kompetensi {misi.kompetensi.kode} belum "
                    "dipetakan ke Game 1 atau Game 2."
                )
            },
            status=400,
        )

    payload = {
        "sub": misi.siswa.kode_samaran,
        "nama": misi.siswa.nama,
        "misi": misi.id,
        "kompetensi": misi.kompetensi.kode,
        "ranah": misi.kompetensi.ranah,
        "level": misi.level,
        "mode": misi.jenis,
        "home": request.build_absolute_uri("/murid/"),
        "logout": request.build_absolute_uri("/logout/"),
        "api": request.build_absolute_uri("/api/game/event/"),
        "exp": int(time.time()) + settings.LENTERA_GAME_TOKEN_TTL,
    }

    token = jwt.encode(
        payload,
        settings.LENTERA_GAME_SECRET,
        algorithm="HS256",
    )

    # Pertahankan query parameter URL yang sudah ada.
    parsed = urlsplit(url_game)

    query = dict(
        parse_qsl(
            parsed.query,
            keep_blank_values=True,
        )
    )

    query["ssoToken"] = token

    tujuan = urlunsplit(
        (
            parsed.scheme,
            parsed.netloc,
            parsed.path,
            urlencode(query),
            parsed.fragment,
        )
    )

    return redirect(tujuan)

def _waktu(value):
    if not value or not isinstance(value, str):
        return None

    try:
        waktu = parse_datetime(value)
    except (ValueError, TypeError):
        return None

    if waktu is None:
        return None

    if timezone.is_naive(waktu):
        return timezone.make_aware(waktu)

    return waktu


def _boolean(value, default=None):
    """Parsing boolean secara ketat."""
    if isinstance(value, bool):
        return value

    if isinstance(value, int) and value in (0, 1):
        return bool(value)

    if isinstance(value, str):
        nilai = value.strip().lower()

        if nilai == "true":
            return True

        if nilai == "false":
            return False

    if value is None:
        return default

    raise ValueError("Nilai boolean tidak valid.")


def _state_siswa(siswa, kompetensi):
    """Mengambil kondisi penguasaan terbaru."""
    penguasaan = Penguasaan.objects.filter(
        siswa=siswa,
        kompetensi=kompetensi,
    ).first()

    if penguasaan is None:
        return {
            "p_l": round(engine.P_AWAL, 3),
            "status": engine.status(engine.P_AWAL),
            "level": 1,
        }

    return {
        "p_l": round(penguasaan.p_l, 3),
        "status": penguasaan.status,
        "level": penguasaan.level,
    }

def _norm(event, misi):
    """
    Menerima event Game 1 dan Game 2.

    Field yang didukung:
    - kompetensi / competency
    - benar / is_correct
    - item_id / question_id
    - pola / error_code / error_pattern
    - waktu_ms / response_time_ms / response_time_seconds
    - petunjuk / used_hint
    """
    if not isinstance(event, dict):
        raise ValueError(
            "Setiap event harus berbentuk objek JSON."
        )

    kode = (
        event.get("kompetensi")
        or event.get("competency")
        or misi.kompetensi.kode
    )

    kode = normalisasi_kompetensi(kode)

    if kode != misi.kompetensi.kode:
        raise ValueError(
            "Kompetensi event tidak cocok dengan kompetensi misi."
        )

    try:
        level = int(event.get("level", misi.level))
    except (TypeError, ValueError):
        raise ValueError("Level soal tidak valid.")

    if level != misi.level or not 1 <= level <= 5:
        raise ValueError(
            "Level event tidak cocok dengan level misi."
        )

    benar = _boolean(
        event.get("benar", event.get("is_correct")),
        default=None,
    )

    if benar is None:
        raise ValueError("Field benar tidak tersedia.")

    item_id = str(
        event.get("item_id")
        or event.get("question_id")
        or ""
    ).strip()

    if len(item_id) > 40:
        raise ValueError("ID soal terlalu panjang.")

    raw_ms = event.get("waktu_ms")

    if raw_ms is None:
        raw_ms = event.get("response_time_ms")

    if raw_ms is None:
        try:
            raw_ms = round(
                float(event.get("response_time_seconds", 0)) * 1000
            )
        except (TypeError, ValueError, OverflowError):
            raw_ms = 0

    try:
        waktu_ms = int(raw_ms)
    except (TypeError, ValueError, OverflowError):
        raise ValueError("Waktu pengerjaan tidak valid.")

    if waktu_ms < 0:
        raise ValueError(
            "Waktu pengerjaan tidak boleh negatif."
        )

    waktu_ms = min(
        waktu_ms,
        24 * 60 * 60 * 1000,
    )

    event_id = str(event.get("event_id") or "").strip()

    if len(event_id) > 64:
        raise ValueError("ID event terlalu panjang.")

    pola = str(
        event.get("pola")
        or event.get("error_code")
        or event.get("error_pattern")
        or ""
    ).strip()

    if len(pola) > 30:
        pola = ""

    definisi_pola = getattr(engine, "POLA", {})

    if pola not in definisi_pola:
        pola = ""
    elif definisi_pola[pola][1] != kode:
        pola = ""

    petunjuk = _boolean(
        event.get("petunjuk", event.get("used_hint")),
        default=False,
    )

    return {
        "kode": kode,
        "benar": benar,
        "level": level,
        "item_id": item_id,
        "pola": pola,
        "waktu_ms": waktu_ms,
        "petunjuk": petunjuk,
        "event_id": event_id,
        "waktu": _waktu(event.get("waktu")),
    }

def _boleh_selesai(misi, events=None):
    """
    Misi selesai ketika jumlah ID soal unik dengan jawaban
    benar yang sudah tersimpan mencapai jumlah wajib.
    """
    jumlah_wajib = _jumlah_soal_wajib(misi)

    if jumlah_wajib <= 0:
        return False

    jumlah_benar = (
        Jawaban.objects.filter(
            misi=misi,
            kompetensi=misi.kompetensi,
            benar=True,
        )
        .exclude(item_id="")
        .values("item_id")
        .distinct()
        .count()
    )

    return jumlah_benar >= jumlah_wajib

@csrf_exempt
@require_POST
def api_event(request):
    if len(request.body) > 256 * 1024:
        return JsonResponse(
            {"error": "Ukuran permintaan terlalu besar."},
            status=413,
        )

    try:
        data = json.loads(request.body or "{}")
    except (json.JSONDecodeError, UnicodeDecodeError):
        return JsonResponse(
            {"error": "JSON permintaan tidak valid."},
            status=400,
        )

    if not isinstance(data, dict):
        return JsonResponse(
            {"error": "Body harus berupa objek JSON."},
            status=400,
        )

    try:
        selesai_diminta = _boolean(
            data.get("selesai"),
            default=False,
        )
    except ValueError:
        return JsonResponse(
            {"error": "Field selesai harus berupa boolean."},
            status=400,
        )

    # Token utama menggunakan Authorization Bearer.
    # Dukungan ssoToken di body dipertahankan untuk kompatibilitas.
    authorization = request.headers.get(
        "Authorization",
        "",
    ).strip()

    if authorization:
        if not authorization.startswith("Bearer "):
            return JsonResponse(
                {"error": "Format Authorization tidak valid."},
                status=401,
            )

        token = authorization[7:].strip()
    else:
        token = str(data.get("ssoToken") or "").strip()

    if not token:
        return JsonResponse(
            {"error": "Token misi tidak tersedia."},
            status=401,
        )

    try:
        claims = jwt.decode(
            token,
            settings.LENTERA_GAME_SECRET,
            algorithms=["HS256"],
            leeway=getattr(
                settings,
                "LENTERA_GAME_EVENT_GRACE",
                0,
            ),
        )

        misi_id = int(claims["misi"])
        kode_token = normalisasi_kompetensi(
            claims["kompetensi"]
        )
        level_token = int(claims["level"])
        sub = str(claims["sub"])

    except (
        jwt.PyJWTError,
        KeyError,
        ValueError,
        TypeError,
    ):
        return JsonResponse(
            {"error": "Token misi tidak valid atau kedaluwarsa."},
            status=401,
        )

    events_raw = data.get("events")

    # Mendukung event tunggal dari Game 1 versi lama.
    if events_raw is None:
        if "item_id" in data or "question_id" in data:
            events_raw = [data]
        else:
            events_raw = []

    if not isinstance(events_raw, list):
        return JsonResponse(
            {"error": "Field events harus berupa array."},
            status=400,
        )

    if len(events_raw) > 20:
        return JsonResponse(
            {"error": "Maksimal 20 event per permintaan."},
            status=400,
        )

    try:
        with transaction.atomic():
            misi = (
                Misi.objects.select_for_update()
                .select_related("siswa", "kompetensi")
                .get(
                    id=misi_id,
                    siswa__kode_samaran=sub,
                )
            )

            if (
                misi.kompetensi.kode != kode_token
                or misi.level != level_token
            ):
                return JsonResponse(
                    {
                        "error": (
                            "Token tidak cocok dengan "
                            "kompetensi atau level misi."
                        )
                    },
                    status=403,
                )

            if _jumlah_soal_wajib(misi) <= 0:
                return JsonResponse(
                    {
                        "error": (
                            "Kompetensi dan level ini belum "
                            "memiliki soal yang tersedia."
                        )
                    },
                    status=409,
                )

            # Semua event divalidasi terlebih dahulu.
            events = [
                _norm(event, misi)
                for event in events_raw
            ]

            if misi.status == "selesai":
                if selesai_diminta and not events:
                    return JsonResponse({
                        "state": _state_siswa(
                            misi.siswa,
                            misi.kompetensi,
                        ),
                        "selesai": True,
                    })

                semua_event_sudah_ada = bool(events) and all(
                    event["event_id"]
                    and Jawaban.objects.filter(
                        event_id=event["event_id"],
                    ).exists()
                    for event in events
                )

                if semua_event_sudah_ada:
                    return JsonResponse({
                        "state": _state_siswa(
                            misi.siswa,
                            misi.kompetensi,
                        ),
                        "selesai": True,
                    })

                return JsonResponse(
                    {"error": "Misi ini sudah selesai."},
                    status=409,
                )

            # Simpan jawaban.
            for event in events:
                try:
                    with transaction.atomic():
                        catat_jawaban(
                            misi.siswa,
                            event["kode"],
                            event["benar"],
                            event["level"],
                            event["item_id"],
                            event["pola"],
                            event["waktu_ms"],
                            event["petunjuk"],
                            event["waktu"],
                            event["event_id"],
                            misi=misi,
                        )
                except IntegrityError:
                    # Event yang sama sudah tersimpan.
                    continue

            state = _state_siswa(
                misi.siswa,
                misi.kompetensi,
            )

            if selesai_diminta:
                if not _boleh_selesai(misi, events):
                    return JsonResponse({
                        "state": state,
                        "selesai": False,
                        "error": (
                            "Belum seluruh soal wajib dijawab "
                            "benar. Misi belum ditandai selesai."
                        ),
                    })

                misi.status = "selesai"
                misi.save(update_fields=["status"])

                if misi.jenis == "diagnostik":
                    selesaikan_diagnostik(misi)
                else:
                    Siswa.objects.filter(
                        id=misi.siswa_id,
                    ).update(
                        poin=F("poin") + 10
                    )

            return JsonResponse({
                "state": _state_siswa(
                    misi.siswa,
                    misi.kompetensi,
                ),
                "selesai": misi.status == "selesai",
            })

    except Misi.DoesNotExist:
        return JsonResponse(
            {"error": "Misi tidak ditemukan."},
            status=404,
        )

    except Kompetensi.DoesNotExist:
        return JsonResponse(
            {"error": "Kode kompetensi tidak dikenal."},
            status=400,
        )

    except (ValueError, TypeError, OverflowError) as error:
        return JsonResponse(
            {"error": str(error)},
            status=400,
        )

KODE_LITERASI = {
    "LT-KOSAKATA",
    "LT-LANCAR",
    "LT-PAHAM",
    "LT-KALIMAT",
}

KODE_NUMERASI = {
    "NM-CERITA",
    "NM-NILAI-TEMPAT",
    "NM-REGROUPING",
    "NM-POLA",
}

KODE_GAME = KODE_LITERASI | KODE_NUMERASI

JUMLAH_SOAL_PER_LEVEL = 5


def jumlah_soal_game(kode, level):
    """
    Menentukan jumlah soal yang dapat digunakan oleh layanan
    dashboard untuk suatu kompetensi dan level.

    Semua kompetensi literasi memakai bank soal Game 1.
    Semua kompetensi numerasi memakai bank soal Game 2.

    Catatan: jumlah ini mengasumsikan Game 1 dan Game 2
    masing-masing memiliki 5 soal pada setiap level.
    """
    try:
        level = int(level)
    except (TypeError, ValueError):
        return 0

    if not 1 <= level <= 5:
        return 0

    if kode in KODE_LITERASI:
        return JUMLAH_SOAL_PER_LEVEL

    if kode in KODE_NUMERASI:
        return JUMLAH_SOAL_PER_LEVEL

    return 0


def level_game_tersedia(
    kode,
    level,
    level_min=1,
    level_maks=5,
):
    """
    Memilih level terdekat yang tersedia dalam rentang aktivitas.

    Mengembalikan None jika kode game tidak dikenal atau
    tidak ada level yang dapat digunakan.
    """
    try:
        level = int(level)
        level_min = max(1, int(level_min))
        level_maks = min(5, int(level_maks))
    except (TypeError, ValueError):
        return None

    if level_min > level_maks:
        return None

    tersedia = [
        nomor
        for nomor in range(level_min, level_maks + 1)
        if jumlah_soal_game(kode, nomor) > 0
    ]

    if not tersedia:
        return None

    return min(
        tersedia,
        key=lambda nomor: (abs(nomor - level), nomor),
    )


def kode_game_tersedia(kode):
    """Memastikan kompetensi telah dipetakan ke Game 1 atau Game 2."""
    return kode in KODE_GAME and kode in getattr(
        settings,
        "LENTERA_GAMES",
        {},
    )

def _langkah(p, benar):
    """Perbarui estimasi penguasaan setelah satu jawaban."""
    p.p_l = engine.update_bkt(p.p_l, benar)

    p.salah_beruntun = (
        0 if benar else p.salah_beruntun + 1
    )

    p.level, p.p_l, p.salah_beruntun = engine.atur_level(
        p.level,
        p.p_l,
        p.salah_beruntun,
    )


def hitung_ulang(siswa, kompetensi):
    """
    Hitung ulang BKT jika jawaban lama baru tersinkronisasi.

    Jawaban asesmen diagnostik tidak dicampur dengan
    jawaban latihan biasa.
    """
    penguasaan, _ = Penguasaan.objects.get_or_create(
        siswa=siswa,
        kompetensi=kompetensi,
        defaults={"p_l": engine.P_AWAL},
    )

    diagnostik = (
        HasilDiagnostik.objects.filter(
            siswa=siswa,
            kompetensi=kompetensi,
        )
        .order_by("dibuat", "id")
        .last()
    )

    if diagnostik:
        penguasaan.p_l = diagnostik.p_awal
        penguasaan.level = diagnostik.level_awal
        penguasaan.salah_beruntun = 0
    else:
        penguasaan.p_l = engine.P_AWAL
        penguasaan.level = 1
        penguasaan.salah_beruntun = 0

    jawaban = Jawaban.objects.filter(
        siswa=siswa,
        kompetensi=kompetensi,
    ).filter(
        Q(misi__isnull=True) | ~Q(misi__jenis="diagnostik")
    )

    if diagnostik:
        jawaban = jawaban.filter(
            dibuat__gte=diagnostik.dibuat
        )

    for item in jawaban.order_by("dibuat", "id"):
        _langkah(penguasaan, item.benar)

        Jawaban.objects.filter(id=item.id).update(
            p_setelah=penguasaan.p_l
        )

    penguasaan.save()

    return penguasaan

@transaction.atomic
def catat_jawaban(
    siswa,
    kode,
    benar,
    level=0,
    item_id="",
    pola="",
    waktu_ms=0,
    petunjuk=False,
    waktu=None,
    event_id="",
    misi=None,
):
    """
    Satu-satunya pintu penyimpanan jawaban belajar.

    Dipanggil oleh API game dan perintah seed demo.
    event_id digunakan agar pengiriman ulang tidak
    menggandakan catatan yang sudah tersimpan.
    """
    if event_id and Jawaban.objects.filter(
        event_id=event_id
    ).exists():
        return None

    kompetensi = Kompetensi.objects.get(kode=kode)

    pola_valid = ""

    if not benar and pola:
        definisi = engine.POLA.get(pola)

        if definisi and definisi[1] == kode:
            pola_valid = pola

    penguasaan, _ = Penguasaan.objects.get_or_create(
        siswa=siswa,
        kompetensi=kompetensi,
        defaults={"p_l": engine.P_AWAL},
    )

    penguasaan = Penguasaan.objects.select_for_update().get(
        id=penguasaan.id
    )

    waktu = waktu or timezone.now()

    diagnostik = bool(
        misi and misi.jenis == "diagnostik"
    )

    terlambat = (
        not diagnostik
        and Jawaban.objects.filter(
            siswa=siswa,
            kompetensi=kompetensi,
            dibuat__gt=waktu,
        )
        .filter(
            Q(misi__isnull=True) | ~Q(misi__jenis="diagnostik")
        )
        .exists()
    )

    jawaban = Jawaban.objects.create(
        siswa=siswa,
        kompetensi=kompetensi,
        misi=misi,
        level=level or penguasaan.level,
        item_id=str(item_id or "")[:40],
        benar=bool(benar),
        pola=pola_valid,
        waktu_ms=max(0, int(waktu_ms or 0)),
        petunjuk=bool(petunjuk),
        event_id=(event_id or None),
        dibuat=waktu,
    )

    # Jawaban diagnostik baru diterapkan setelah asesmen selesai.
    if diagnostik:
        return {
            "p_l": round(penguasaan.p_l, 3),
            "status": penguasaan.status,
            "level": penguasaan.level,
        }

    if terlambat:
        penguasaan = hitung_ulang(
            siswa,
            kompetensi,
        )
    else:
        _langkah(penguasaan, bool(benar))

        jawaban.p_setelah = penguasaan.p_l
        jawaban.save(update_fields=["p_setelah"])

        penguasaan.save()

    if benar:
        Siswa.objects.filter(id=siswa.id).update(
            poin=F("poin") + 1
        )

    return {
        "p_l": round(penguasaan.p_l, 3),
        "status": penguasaan.status,
        "level": penguasaan.level,
    }

def misi_aktif(siswa):
    """
    Memilih misi aktif per ranah.

    Hanya memilih kompetensi yang mempunyai jalur ke Game 1
    atau Game 2. Tidak membuat misi cadangan yang tidak dapat
    dimainkan.
    """
    penguasaan_lengkap(siswa)

    hasil = []

    game_codes = {
        kode
        for kode in getattr(settings, "LENTERA_GAMES", {})
        if kode in KODE_GAME
    }

    for ranah in ("literasi", "numerasi"):
        misi_terpilih = None

        calon_misi = (
            siswa.misi.filter(
                status="baru",
                kompetensi__ranah=ranah,
                kompetensi__kode__in=game_codes,
            )
            .select_related("kompetensi", "aktivitas")
            .order_by("-dari_guru", "-id")
        )

        # Prioritaskan misi yang sudah ditugaskan.
        for calon in calon_misi:
            level_min = (
                calon.aktivitas.level_min
                if calon.aktivitas else 1
            )
            level_maks = (
                calon.aktivitas.level_maks
                if calon.aktivitas else 5
            )

            level_valid = level_game_tersedia(
                calon.kompetensi.kode,
                calon.level,
                level_min,
                level_maks,
            )

            if level_valid is None:
                continue

            if calon.level != level_valid:
                calon.level = level_valid
                calon.save(update_fields=["level"])

            misi_terpilih = calon
            break

        # Jika belum ada, pilih kompetensi dengan penguasaan
        # terendah yang telah dipetakan ke game.
        if misi_terpilih is None:
            calon_penguasaan = (
                siswa.penguasaan.filter(
                    kompetensi__ranah=ranah,
                    kompetensi__kode__in=game_codes,
                )
                .select_related("kompetensi")
                .order_by("p_l", "kompetensi__urutan")
            )

            for penguasaan in calon_penguasaan:
                level_valid = level_game_tersedia(
                    penguasaan.kompetensi.kode,
                    penguasaan.level,
                )

                if level_valid is None:
                    continue

                misi_terpilih = Misi.objects.create(
                    siswa=siswa,
                    kompetensi=penguasaan.kompetensi,
                    level=level_valid,
                    jenis="misi",
                )
                break

        if misi_terpilih is not None:
            hasil.append(misi_terpilih)

    return hasil

@transaction.atomic
def terapkan_diagnostik(
    siswa,
    kompetensi,
    benar,
    total,
    sumber="manual",
):
    """Terapkan hasil diagnostik menjadi state awal siswa."""
    benar = int(benar)
    total = int(total)

    if total <= 0:
        raise ValueError(
            "Jumlah soal diagnostik harus lebih dari nol."
        )

    if benar < 0 or benar > total:
        raise ValueError(
            "Jumlah benar harus berada antara nol dan total."
        )

    if benar:
        Siswa.objects.filter(id=siswa.id).update(
            poin=F("poin") + 1
        )

    level_awal, p_awal = engine.level_dari_skor(
        benar,
        total,
    )

    hasil = HasilDiagnostik.objects.create(
        siswa=siswa,
        kompetensi=kompetensi,
        benar=benar,
        total=total,
        level_awal=level_awal,
        p_awal=p_awal,
        sumber=sumber,
    )

    penguasaan, _ = Penguasaan.objects.get_or_create(
        siswa=siswa,
        kompetensi=kompetensi,
        defaults={"p_l": p_awal},
    )

    penguasaan.p_l = p_awal
    penguasaan.level = level_awal
    penguasaan.salah_beruntun = 0
    penguasaan.save()

    return hasil


@transaction.atomic
def aktifkan_diagnostik(kelas):
    """
    Membuat misi diagnostik untuk kompetensi yang
    terhubung ke Game 1 atau Game 2.
    """
    jumlah = 0

    game_codes = {
        kode
        for kode in getattr(settings, "LENTERA_GAMES", {})
        if kode in KODE_GAME
    }

    kompetensi_tersedia = Kompetensi.objects.filter(
        kode__in=game_codes
    )

    for siswa in kelas.siswa.all():
        for kompetensi in kompetensi_tersedia:
            level_awal = level_game_tersedia(
                kompetensi.kode,
                1,
            )

            if level_awal is None:
                continue

            sudah_selesai = HasilDiagnostik.objects.filter(
                siswa=siswa,
                kompetensi=kompetensi,
            ).exists()

            masih_aktif = Misi.objects.filter(
                siswa=siswa,
                kompetensi=kompetensi,
                jenis="diagnostik",
                status="baru",
            ).exists()

            if sudah_selesai or masih_aktif:
                continue

            Misi.objects.create(
                siswa=siswa,
                kompetensi=kompetensi,
                level=level_awal,
                jenis="diagnostik",
                dari_guru=True,
            )

            jumlah += 1

    return jumlah


@transaction.atomic
def selesaikan_diagnostik(misi):
    """
    Hitung hasil diagnostik berdasarkan jawaban pertama
    untuk setiap ID soal.
    """
    jawaban_pertama = {}

    for jawaban in (
        Jawaban.objects.filter(misi=misi)
        .select_related("kompetensi")
        .order_by("dibuat", "id")
    ):
        if not jawaban.item_id:
            continue

        key = (
            jawaban.kompetensi_id,
            jawaban.item_id,
        )

        jawaban_pertama.setdefault(key, jawaban)

    per_kompetensi = {}

    for (kompetensi_id, _item_id), jawaban in (
        jawaban_pertama.items()
    ):
        statistik = per_kompetensi.setdefault(
            kompetensi_id,
            {
                "kompetensi": jawaban.kompetensi,
                "benar": 0,
                "total": 0,
            },
        )

        statistik["total"] += 1

        if jawaban.benar:
            statistik["benar"] += 1

    hasil = []

    for statistik in per_kompetensi.values():
        hasil.append(
            terapkan_diagnostik(
                siswa=misi.siswa,
                kompetensi=statistik["kompetensi"],
                benar=statistik["benar"],
                total=statistik["total"],
                sumber="game",
            )
        )

    return hasil


def status_diagnostik(kelas):
    """Menyiapkan status asesmen diagnostik setiap siswa."""
    kompetensi = list(Kompetensi.objects.all())

    siswa = list(
        kelas.siswa.select_related("user")
    )

    hasil_db = (
        HasilDiagnostik.objects.filter(siswa__in=siswa)
        .order_by("dibuat", "id")
    )

    hasil = {}

    for item in hasil_db:
        hasil[(item.siswa_id, item.kompetensi_id)] = item

    menunggu = set(
        Misi.objects.filter(
            siswa__in=siswa,
            jenis="diagnostik",
            status="baru",
        ).values_list(
            "siswa_id",
            "kompetensi_id",
        )
    )

    baris = []

    for murid in siswa:
        pilihan = []

        for kompetensi_item in kompetensi:
            key = (
                murid.id,
                kompetensi_item.id,
            )

            pilihan.append({
                "h": hasil.get(key),
                "tunggu": key in menunggu,
            })

        baris.append({
            "siswa": murid,
            "sel": pilihan,
        })

    jumlah_selesai = sum(
        1
        for baris_item in baris
        for pilihan in baris_item["sel"]
        if pilihan["h"]
    )

    return {
        "kompetensi": kompetensi,
        "baris": baris,
        "selesai": jumlah_selesai,
        "total": len(siswa) * len(kompetensi),
        "main": sorted(
            kode
            for kode in settings.LENTERA_GAMES
            if kode in KODE_GAME
        ),
    }

@transaction.atomic
def tugaskan_aktivitas(
    guru,
    kelas,
    aktivitas,
    target,
    ids=(),
):
    """
    Menugaskan aktivitas ke seluruh kelas, kelompok kemampuan,
    atau siswa terpilih. Aktivitas harus terhubung ke game.
    """
    siswa_kelas = list(kelas.siswa.all())

    kode = aktivitas.kompetensi.kode
    level_min = max(1, int(aktivitas.level_min))
    level_maks = min(5, int(aktivitas.level_maks))

    dapat_dimainkan = (
        kode_game_tersedia(kode)
        and level_game_tersedia(
            kode,
            level_min,
            level_min,
            level_maks,
        ) is not None
    )

    penguasaan = {
        p.siswa_id: p
        for p in Penguasaan.objects.filter(
            kompetensi=aktivitas.kompetensi,
            siswa__in=siswa_kelas,
        )
    }

    def status_siswa(siswa):
        p = penguasaan.get(siswa.id)

        return engine.status(
            p.p_l if p else engine.P_AWAL
        )

    if target in ("Mulai", "Berkembang", "Cakap"):
        dipilih = [
            siswa
            for siswa in siswa_kelas
            if status_siswa(siswa) == target
        ]

        label = f"Kelompok {target}"

    elif target == "pilih":
        id_terpilih = {str(item) for item in ids}

        dipilih = [
            siswa
            for siswa in siswa_kelas
            if str(siswa.id) in id_terpilih
        ]

        label = "Siswa terpilih"

    else:
        dipilih = siswa_kelas
        label = "Seluruh kelas"

    jumlah_dibuat = 0

    if dapat_dimainkan:
        for siswa in dipilih:
            p = penguasaan.get(siswa.id)
            level_siswa = p.level if p else level_min

            level = level_game_tersedia(
                kode,
                level_siswa,
                level_min,
                level_maks,
            )

            if level is None:
                continue

            misi = (
                Misi.objects.filter(
                    siswa=siswa,
                    kompetensi=aktivitas.kompetensi,
                    status="baru",
                    jenis="misi",
                )
                .order_by("id")
                .first()
            )

            if misi:
                misi.level = level
                misi.dari_guru = True
                misi.aktivitas = aktivitas

                misi.save(
                    update_fields=[
                        "level",
                        "dari_guru",
                        "aktivitas",
                    ]
                )

            else:
                Misi.objects.create(
                    siswa=siswa,
                    kompetensi=aktivitas.kompetensi,
                    level=level,
                    jenis="misi",
                    dari_guru=True,
                    aktivitas=aktivitas,
                )

            jumlah_dibuat += 1

    Penugasan.objects.create(
        aktivitas=aktivitas,
        kelas=kelas,
        target=label,
        jumlah=jumlah_dibuat,
        dibuat_oleh=guru,
    )

    return jumlah_dibuat, dapat_dimainkan


@transaction.atomic
def terapkan_ke_kelompok(guru, kelas, kompetensi):
    """
    Menerapkan rekomendasi ke siswa berdasarkan penguasaan.

    Hanya membuat misi jika kompetensi dipetakan ke Game 1
    atau Game 2.
    """
    kode = kompetensi.kode

    if not kode_game_tersedia(kode):
        return 0

    jumlah_misi = 0

    for siswa in kelas.siswa.all():
        penguasaan, _ = Penguasaan.objects.get_or_create(
            siswa=siswa,
            kompetensi=kompetensi,
            defaults={"p_l": engine.P_AWAL},
        )

        level = penguasaan.level

        if penguasaan.status == "Cakap":
            level = min(5, level + 1)

        level_valid = level_game_tersedia(
            kode,
            level,
        )

        if level_valid is None:
            continue

        misi = (
            Misi.objects.filter(
                siswa=siswa,
                kompetensi=kompetensi,
                status="baru",
                jenis="misi",
            )
            .order_by("id")
            .first()
        )

        if misi:
            misi.level = level_valid
            misi.dari_guru = True

            misi.save(
                update_fields=[
                    "level",
                    "dari_guru",
                ]
            )

        else:
            Misi.objects.create(
                siswa=siswa,
                kompetensi=kompetensi,
                level=level_valid,
                jenis="misi",
                dari_guru=True,
            )

        jumlah_misi += 1

    return jumlah_misi

@transaction.atomic
def tambah_siswa(kelas, nama):
    """Buat akun murid tanpa kata sandi dan ID samaran unik."""
    dasar = slugify(nama)[:16] or "murid"
    nomor = 0

    while True:
        username = (
            f"{dasar}.{kelas.id}"
            + (str(nomor) if nomor else "")
        )

        if not User.objects.filter(username=username).exists():
            break

        nomor += 1

    user = User(
        username=username,
        first_name=nama[:30],
    )

    user.set_unusable_password()
    user.save()

    Profile.objects.create(
        user=user,
        role="murid",
    )

    while True:
        kode_samaran = f"S-{secrets.token_hex(3).upper()}"

        if not Siswa.objects.filter(
            kode_samaran=kode_samaran
        ).exists():
            break

    avatar_terpakai = set(
        kelas.siswa.values_list(
            "avatar",
            flat=True,
        )
    )

    avatar = next(
        (
            item
            for item in AVATAR
            if item not in avatar_terpakai
        ),
        AVATAR[kelas.siswa.count() % len(AVATAR)],
    )

    siswa = Siswa.objects.create(
        user=user,
        kelas=kelas,
        avatar=avatar,
        kode_samaran=kode_samaran,
    )

    penguasaan_lengkap(siswa)

    return siswa

def buat_guru(username, nama, sandi, kelas=None):
    """
    Membuat akun guru baru untuk Admin: Kelola Guru.
    """
    username = (username or "").strip()

    if not username:
        raise ValueError("Username wajib diisi.")

    if len(sandi or "") < 6:
        raise ValueError("Kata sandi minimal 6 karakter.")

    if User.objects.filter(username=username).exists():
        raise ValueError("Username itu sudah dipakai.")

    user = User.objects.create_user(
        username,
        password=sandi,
        first_name=(nama or "").strip()[:30],
    )

    Profile.objects.create(
        user=user,
        role="guru",
    )

    if kelas is not None:
        tetapkan_wali_kelas(user, kelas)

    return user

def _rel(waktu):
    """Format waktu relatif untuk dashboard guru."""
    if not waktu:
        return ""

    menit = int(
        (timezone.now() - waktu).total_seconds() // 60
    )

    if menit < 5:
        return "baru saja"

    if menit < 60:
        return f"{menit} mnt lalu"

    if menit < 1440:
        return f"{menit // 60} jam lalu"

    return f"{menit // 1440} hari lalu"


def ringkasan_kelas(kelas):
    """
    Menyediakan data untuk Insight Guru, Peta Kompetensi,
    rekomendasi tindak lanjut, dan laporan kelas.

    Aktivitas dan pola kesalahan: 7 hari terakhir.
    Status kompetensi: kondisi terkini.
    Perbandingan kemajuan: kondisi sekarang dengan data
    terakhir sebelum periode 7 hari.
    """
    awal = timezone.now() - timedelta(days=7)

    kompetensi = list(Kompetensi.objects.all())
    siswa = list(kelas.siswa.select_related("user"))

    penguasaan = {
        (p.siswa_id, p.kompetensi_id): p
        for p in Penguasaan.objects.filter(siswa__in=siswa)
    }

    skor_sekarang = {
        kunci: p.p_l
        for kunci, p in penguasaan.items()
    }

    skor_sebelum = {}

    riwayat_sebelum = (
        Jawaban.objects
        .filter(
            siswa__in=siswa,
            dibuat__lt=awal,
            p_setelah__isnull=False,
        )
        .order_by("dibuat", "id")
        .values_list(
            "siswa_id",
            "kompetensi_id",
            "p_setelah",
        )
    )

    for siswa_id, kompetensi_id, nilai in riwayat_sebelum:
        skor_sebelum[(siswa_id, kompetensi_id)] = nilai

    jawaban_7hari = Jawaban.objects.filter(
        siswa__in=siswa,
        dibuat__gte=awal,
    )

    statistik_siswa = {
        item["siswa_id"]: item
        for item in (
            jawaban_7hari
            .values("siswa_id")
            .annotate(
                jumlah=Count("id"),
                benar=Count(
                    "id",
                    filter=Q(benar=True),
                ),
            )
        )
    }

    aktivitas_terakhir = dict(
        Jawaban.objects
        .filter(siswa__in=siswa)
        .values_list("siswa_id")
        .annotate(waktu_terakhir=Max("dibuat"))
    )

    pola_siswa = {}

    pola_per_siswa = (
        jawaban_7hari
        .filter(benar=False)
        .exclude(pola="")
        .values("siswa_id", "pola")
        .annotate(jumlah=Count("id"))
        .order_by("siswa_id", "-jumlah", "pola")
    )

    for item in pola_per_siswa:
        pola_siswa.setdefault(
            item["siswa_id"],
            item["pola"],
        )

    baris = []

    for murid in siswa:
        sel = []

        for k in kompetensi:
            p = penguasaan.get((murid.id, k.id))
            nilai = p.p_l if p else engine.P_AWAL

            sel.append({
                "pct": round(nilai * 100),
                "st": engine.status(nilai),
                "level": p.level if p else 1,
            })

        statistik = statistik_siswa.get(
            murid.id,
            {"jumlah": 0, "benar": 0},
        )

        jumlah = statistik["jumlah"]
        benar = statistik["benar"]
        pola = pola_siswa.get(murid.id, "")

        baris.append({
            "siswa": murid,
            "sel": sel,
            "perhatian": (
                sum(
                    item["st"] == "Mulai"
                    for item in sel
                ) >= kelas.ambang_perhatian
            ),
            "n_jawab": jumlah,
            "akurasi": (
                round(100 * benar / jumlah)
                if jumlah else ""
            ),
            "terakhir": _rel(
                aktivitas_terakhir.get(murid.id)
            ),
            "pola": (
                engine.POLA[pola][0]
                if pola in engine.POLA
                else ""
            ),
            "pola_saran": (
                engine.POLA[pola][2]
                if pola in engine.POLA
                else ""
            ),
        })

    def rata_ranah(data, ranah):
        nilai = [
            (
                data.get((murid.id, k.id), engine.P_AWAL)
                * 100
            )
            for murid in siswa
            for k in kompetensi
            if k.ranah == ranah
        ]

        return sum(nilai) / len(nilai) if nilai else 0

    ranah = []

    for nama_ranah in ("literasi", "numerasi"):
        sekarang = rata_ranah(
            skor_sekarang,
            nama_ranah,
        )
        sebelumnya = rata_ranah(
            skor_sebelum,
            nama_ranah,
        )

        ranah.append({
            "kode": nama_ranah,
            "nama": nama_ranah.capitalize(),
            "nilai": round(sekarang),
            "delta": abs(round(sekarang - sebelumnya)),
            "naik": sekarang >= sebelumnya,
        })

    total_waktu = jawaban_7hari.aggregate(
        total=Sum("waktu_ms")
    )["total"] or 0

    waktu = round(
        total_waktu / 60000 / max(len(siswa), 1)
    )

    kesulitan = (
        jawaban_7hari
        .filter(benar=False)
        .exclude(pola="")
        .values("pola")
        .annotate(
            jumlah_siswa=Count("siswa", distinct=True)
        )
        .order_by("-jumlah_siswa", "pola")[:3]
    )

    daftar_kesulitan = [
        {
            "label": engine.POLA[item["pola"]][0],
            "n": item["jumlah_siswa"],
            "saran": engine.POLA[item["pola"]][2],
        }
        for item in kesulitan
        if item["pola"] in engine.POLA
    ]

    rekomendasi = None

    if siswa and kompetensi:
        rata_kompetensi = [
            (
                sum(
                    data["sel"][i]["pct"]
                    for data in baris
                ) / len(baris)
            )
            for i in range(len(kompetensi))
        ]

        indeks_terendah = min(
            range(len(kompetensi)),
            key=lambda i: rata_kompetensi[i],
        )

        k = kompetensi[indeks_terendah]
        kelompok = []

        for status in ("Mulai", "Berkembang", "Cakap"):
            anggota = [
                data
                for data in baris
                if data["sel"][indeks_terendah]["st"] == status
            ]

            level = (
                round(
                    sum(
                        data["sel"][indeks_terendah]["level"]
                        for data in anggota
                    ) / len(anggota)
                )
                if anggota else 1
            )

            aktivitas = list(
                Aktivitas.objects.filter(
                    kompetensi=k,
                    tervalidasi=True,
                    level_min__lte=level,
                    level_maks__gte=level,
                )[:2]
            )

            kelompok.append({
                "status": status,
                "n": len(anggota),
                "teks": engine.TINDAK[k.ranah][status],
                "aktivitas": aktivitas,
            })

        rekomendasi = {
            "kompetensi": k,
            "kelompok": kelompok,
        }

    return {
        "kelas": kelas,
        "kompetensi": kompetensi,
        "baris": baris,
        "n": len(siswa),
        "ranah": ranah,
        "waktu": waktu,
        "rek": rekomendasi,
        "perhatian": sum(
            item["perhatian"]
            for item in baris
        ),
        "kesulitan": daftar_kesulitan,
    }

def _kartu(m):

def _jalur(pen):
    """Jalur Lenteraku: kurva mulus melewati 8 lentera; Cakap = menyala, kompetensi pertama yang belum Cakap = 'kini'."""
    n = len(pen)
    pts = [(22 + i * 256 / max(n - 1, 1), 160 - i * 124 / max(n - 1, 1) + (8 if i % 2 else -8)) for i in range(n)]
    d = f"M{pts[0][0]:.0f} {pts[0][1]:.0f}"
    for i in range(n - 1):
        p0, p1, p2, p3 = pts[max(i - 1, 0)], pts[i], pts[i + 1], pts[min(i + 2, n - 1)]
        d += (f" C{p1[0] + (p2[0] - p0[0]) / 6:.0f} {p1[1] + (p2[1] - p0[1]) / 6:.0f} "
              f"{p2[0] - (p3[0] - p1[0]) / 6:.0f} {p2[1] - (p3[1] - p1[1]) / 6:.0f} {p2[0]:.0f} {p2[1]:.0f}")
    kini = next((i for i, p in enumerate(pen) if p.status != "Cakap"), -1)
    nodes = [{"x": round(x), "y": round(y), "nama": p.kompetensi.nama, "st": "done" if p.status == "Cakap" else "kini" if i == kini else "next"}
             for i, ((x, y), p) in enumerate(zip(pts, pen))]
    return {"d": d, "nodes": nodes}

def guru_view(fn):
    """Khusus guru. Menyuntikkan kelas wali; guru yang belum punya kelas diarahkan ke Pengaturan."""
    @wraps(fn)
    @role_required("guru")
    def wrapped(request, *a, **kw):
        kelas = Kelas.objects.filter(wali_kelas=request.user).first()
        if not kelas:
            messages.error(request, "Buat kelas dulu di menu Pengaturan.")
            return redirect("/guru/pengaturan/")
        return fn(request, kelas, *a, **kw)
    return wrapped

@guru_view
def guru_aktivitas(request, kelas):
    if request.method == "POST":
        aksi, p = request.POST.get("aksi"), request.POST
        try:
            if aksi == "tambah":
                a, b = sorted((int(p["level_min"]), int(p["level_maks"])))
                if not p["judul"].strip():
                    raise ValueError
                Aktivitas.objects.create(judul=p["judul"].strip()[:80], kompetensi=_kompetensi_game(p["kompetensi"]),
                                         level_min=max(1, a), level_maks=min(5, b), menit=int(p.get("menit") or 10), deskripsi=p.get("deskripsi", ""), dibuat_oleh=request.user)
                messages.success(request, "Aktivitas ditambahkan. Validasi dulu sebelum bisa ditugaskan.")
            elif aksi == "edit":
                obj = get_object_or_404(Aktivitas, id=p["id"])
                a, b = sorted((int(p["level_min"]), int(p["level_maks"])))
                if not p["judul"].strip():
                    raise ValueError
                obj.judul = p["judul"].strip()[:80]
                obj.kompetensi = _kompetensi_game(p["kompetensi"])
                obj.level_min, obj.level_maks = max(1, a), min(5, b)
                obj.menit = int(p.get("menit") or 10)
                obj.deskripsi = p.get("deskripsi", "")
                obj.save()
                messages.success(request, f"“{obj.judul}” diperbarui.")
            elif aksi == "validasi":
                a = get_object_or_404(Aktivitas, id=p["id"])
                a.tervalidasi = not a.tervalidasi
                a.save()
                messages.success(request, f"“{a.judul}” {'divalidasi' if a.tervalidasi else 'dicabut validasinya'}.")
            elif aksi == "hapus":
                a = get_object_or_404(Aktivitas, id=p["id"])
                judul = a.judul
                a.delete()
                messages.success(request, f"“{judul}” dihapus dari bank aktivitas.")
            elif aksi == "tugaskan":
                a = get_object_or_404(Aktivitas, id=p["id"])
                if not a.tervalidasi:
                    messages.error(request, "Aktivitas belum divalidasi, jadi belum bisa ditugaskan.")
                else:
                    n, main = tugaskan_aktivitas(request.user, kelas, a, p.get("target", "kelas"), request.POST.getlist("siswa"))
                    messages.success(request, (f"Misi “{a.judul}” dikirim ke {n} siswa." if main else f"Penugasan “{a.judul}” untuk {n} siswa dicatat.") if n else "Tidak ada siswa pada target itu.")
            elif aksi == "selesai_penugasan":
                t = get_object_or_404(Penugasan, id=p["id"], kelas=kelas)
                t.selesai = not t.selesai
                t.save()
        except (ValueError, KeyError):
            messages.error(request, "Isian tidak valid.")
        return redirect(request.get_full_path())
    f, daftar = {k: request.GET.get(k, "") for k in ("kompetensi", "level")}, Aktivitas.objects.select_related("kompetensi")
    if f["kompetensi"]:
        daftar = daftar.filter(kompetensi__kode=f["kompetensi"])
    if f["level"].isdigit():
        daftar = daftar.filter(level_min__lte=int(f["level"]), level_maks__gte=int(f["level"]))
    daftar = list(daftar)
    for a in daftar:
        a.main = a.kompetensi.kode in settings.LENTERA_GAMES
    return render(request, "belajar/guru_aktivitas.html", _ctx(request, kelas, "aktivitas", daftar=daftar, f=f, kompetensi=Kompetensi.objects.filter(kode__in=settings.LENTERA_GAMES),
                  siswa=kelas.siswa.select_related("user"), penugasan=Penugasan.objects.filter(kelas=kelas).select_related("aktivitas")[:10]))

        if request.POST.get("aksi") == "aktifkan":
            n = aktifkan_diagnostik(kelas)
            messages.success(request, f"{n} misi asesmen dikirim ke murid." if n else "Semua murid sudah punya hasil atau misi asesmen.")
        elif request.POST.get("aksi") == "manual":
            s, n = get_object_or_404(kelas.siswa.all(), id=request.POST["siswa"]), 0
            for k in Kompetensi.objects.all():
                if not request.POST.get(f"b_{k.kode}", "").strip():
                    continue  # kolom "benar" kosong = kompetensi ini tidak diasesmen
                try:
                    b, t = int(request.POST[f"b_{k.kode}"]), int(request.POST.get(f"t_{k.kode}") or 0)
                except ValueError:
                    continue
                if t > 0 and 0 <= b <= t:
                    terapkan_diagnostik(s, k, b, t, "manual")
                    n += 1
            messages.success(request, f"Hasil asesmen {s.nama}: {n} kompetensi disimpan, level awal diperbarui.")
        return redirect("/guru/diagnostik/")
    return render(request, "belajar/guru_diagnostik.html", _ctx(request, kelas, "diagnostik", d=status_diagnostik(kelas), siswa=kelas.siswa.select_related("user")))

def ekspor_csv(kelas, jenis):
    """Unduhan CSV: peta kompetensi, log jawaban, atau hasil diagnostik satu kelas."""
    nama_file = {
        "peta": "lentera_peta_kompetensi.csv",
        "jawaban": "lentera_log_jawaban_30_hari.csv",
        "diagnostik": "lentera_hasil_diagnostik.csv",
    }

    # BOM membantu Excel mengenali karakter UTF-8, termasuk nama Indonesia.
    response = HttpResponse(
        content_type="text/csv; charset=utf-8",
    )
    response["Content-Disposition"] = (
        f'attachment; filename="{nama_file[jenis]}"'
    )
    response.write("\ufeff")

    writer = csv.writer(response, lineterminator="\n")

    # -----------------------------------------
    # 1. EKSPOR PETA KOMPETENSI PER SISWA
    # -----------------------------------------
    if jenis == "peta":
        data = ringkasan_kelas(kelas)

        writer.writerow([
            "Nama siswa",
            *[
                f"{kompetensi.nama} (%)"
                for kompetensi in data["kompetensi"]
            ],
            "Jawaban 7 hari",
            "Akurasi 7 hari (%)",
            "Aktivitas terakhir",
            "Pola kesalahan 7 hari",
        ])

        for baris in data["baris"]:
            writer.writerow([
                baris["siswa"].nama,
                *[
                    sel["pct"]
                    for sel in baris["sel"]
                ],
                baris["n_jawab"],
                baris["akurasi"],
                baris["terakhir"],
                baris["pola"],
            ])

    # -----------------------------------------
    # 2. EKSPOR LOG JAWABAN 30 HARI
    # -----------------------------------------
    elif jenis == "jawaban":
        batas_waktu = timezone.now() - timedelta(days=30)

        writer.writerow([
            "Waktu",
            "Siswa",
            "Kompetensi",
            "Level",
            "ID soal",
            "Jawaban benar",
            "Pola kesalahan",
            "Saran tindak lanjut",
            "Waktu respons (detik)",
            "Menggunakan petunjuk",
            "Jenis misi",
        ])

        jawaban = (
            Jawaban.objects
            .filter(
                siswa__kelas=kelas,
                dibuat__gte=batas_waktu,
            )
            .select_related(
                "siswa__user",
                "kompetensi",
                "misi",
            )
            .order_by("dibuat", "id")[:20000]
        )

        for item in jawaban:
            waktu_lokal = timezone.localtime(item.dibuat)

            pola = (
                engine.POLA[item.pola][0]
                if item.pola in engine.POLA
                else ""
            )

            saran = (
                engine.POLA[item.pola][2]
                if item.pola in engine.POLA
                else ""
            )

            writer.writerow([
                waktu_lokal.strftime("%Y-%m-%d %H:%M:%S"),
                item.siswa.nama,
                item.kompetensi.nama,
                item.level,
                item.item_id,
                "Ya" if item.benar else "Tidak",
                pola,
                saran,
                round(item.waktu_ms / 1000, 1),
                "Ya" if item.petunjuk else "Tidak",
                item.misi.jenis if item.misi else "misi",
            ])

    # -----------------------------------------
    # 3. EKSPOR HASIL ASESMEN DIAGNOSTIK
    # -----------------------------------------
    elif jenis == "diagnostik":
        writer.writerow([
            "Nama siswa",
            "Kompetensi",
            "Jawaban benar",
            "Total soal",
            "Level awal",
            "P(L) awal",
            "Sumber asesmen",
            "Tanggal",
        ])

        hasil = (
            HasilDiagnostik.objects
            .filter(siswa__kelas=kelas)
            .select_related(
                "siswa__user",
                "kompetensi",
            )
            .order_by(
                "siswa__user__first_name",
                "kompetensi__urutan",
                "-dibuat",
            )
        )

        for item in hasil:
            tanggal = timezone.localtime(item.dibuat)

            writer.writerow([
                item.siswa.nama,
                item.kompetensi.nama,
                item.benar,
                item.total,
                item.level_awal,
                item.p_awal,
                item.get_sumber_display(),
                tanggal.strftime("%Y-%m-%d"),
            ])

    return response

@role_required("guru")
def guru_pengaturan(request):
    kelas = Kelas.objects.filter(wali_kelas=request.user).first()
    if request.method == "POST":
        aksi, p = request.POST.get("aksi"), request.POST
        try:
            if aksi == "buat_kelas" and not kelas:
                buat_kelas(request.user, p["nama"].strip()[:20])
                messages.success(request, "Kelas dibuat. Sekarang tambahkan siswa.")
            elif aksi == "tambah_siswa" and kelas:
                nama = [x.strip()[:40] for x in p["nama"].splitlines() if x.strip()]
                for x in nama:
                    tambah_siswa(kelas, x)
                messages.success(request, f"{len(nama)} siswa ditambahkan. Cetak kartu masuknya di bawah.")
            elif aksi == "hapus_siswa" and kelas:
                get_object_or_404(kelas.siswa.all(), id=p["id"]).user.delete()
                messages.success(request, "Siswa dihapus beserta datanya.")
            elif aksi == "ganti_kode" and kelas:
                kelas.kode = kode_kelas_baru(kelas.nama)
                kelas.save()
                messages.success(request, f"Kode kelas baru: {kelas.kode}. Kartu masuk lama perlu dicetak ulang.")
            elif aksi == "ambang" and kelas:
                kelas.ambang_perhatian = max(1, min(8, int(p["ambang"])))
                kelas.save()
                messages.success(request, "Ambang “Perlu perhatian” disimpan.")
            elif aksi == "sandi":
                if len(p["sandi"]) < 6:
                    raise ValueError
                request.user.set_password(p["sandi"])
                request.user.save()
                update_session_auth_hash(request, request.user)
                messages.success(request, "Kata sandi diganti.")
        except IntegrityError:
            messages.error(request, "Nama kelas itu sudah dipakai.")
        except (ValueError, KeyError):
            messages.error(request, "Isian tidak valid (kata sandi minimal 6 karakter).")
        return redirect("/guru/pengaturan/")
    return render(request, "belajar/guru_pengaturan.html", _ctx(request, kelas, "pengaturan", siswa=kelas.siswa.select_related("user") if kelas else []))

// Item Selector LENTERA (port JS dari belajar/engine.py -> pilih_item). Berjalan offline di perangkat murid.
const G = 0.2, S = 0.1;
const peluangBenar = (p, lvSiswa, lvItem) => Math.min(0.97, Math.max(0.03, p * (1 - S) + (1 - p) * G - 0.15 * (lvItem - lvSiswa)));
// kandidat = [{id, level}], riwayat = id soal yang sudah muncul. Zona tantangan: peluang benar terdekat 70%.
export function pilihItem(kandidat, pL, level, riwayat = []) {
  const pool = kandidat.filter((k) => !riwayat.slice(-5).includes(k.id));
  return (pool.length ? pool : kandidat).reduce((a, b) =>
    Math.abs(peluangBenar(pL, level, b.level) - 0.7) < Math.abs(peluangBenar(pL, level, a.level) - 0.7) ? b : a);
}
